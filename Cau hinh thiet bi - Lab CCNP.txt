!=====================================================================
! THIẾT BỊ: Google-R1  (Router 2911)
!=====================================================================
enable
configure terminal
!
hostname Google-R1
interface GigabitEthernet0/0
 description WAN_TO_ISP-R4
 ip address 8.8.8.2 255.255.255.0
 ip nat outside
!
interface GigabitEthernet0/1
 description LAN_TO_SERVERS
 ip address 192.168.0.1 255.255.255.0
 ip nat inside
!
interface GigabitEthernet0/2
 no ip address
 shutdown
!
ip nat inside source static udp 192.168.0.10 53 8.8.8.8 53
ip nat inside source static tcp 192.168.0.10 53 8.8.8.8 53
ip nat inside source static tcp 192.168.0.20 25 8.8.8.9 25
ip nat inside source static tcp 192.168.0.20 110 8.8.8.9 110
ip nat inside source static tcp 192.168.0.30 443 8.8.8.9 443
ip route 0.0.0.0 0.0.0.0 8.8.8.1
end
write memory

!=====================================================================
! THIẾT BỊ: BQP-HN-R1  (Router 2911 - biên, uplink ISP-R1)
!=====================================================================
! LƯU Ý: Trên 2911 phải bật license securityk9 mới dùng được lệnh crypto.
! Chạy lệnh "license boot module ..." bên dưới -> end -> write -> reload, sau đó mới dán phần còn lại.
enable
configure terminal
!
hostname BQP-HN-R1
license boot module c2900 technology-package securityk9
crypto isakmp policy 10
 encr aes 256
 authentication pre-share
 group 5
!
crypto isakmp key BQP@lamlam123 address 201.165.4.2
!
crypto ipsec transform-set TS esp-aes 256 esp-sha-hmac
!
crypto map VPN 10 ipsec-isakmp
 set peer 201.165.4.2
 set transform-set TS
 match address 110
!
interface Loopback0
 ip address 10.1.0.1 255.255.255.255
!
interface Tunnel0
 ip address 10.255.255.1 255.255.255.252
 mtu 1476
 tunnel source GigabitEthernet0/0
 tunnel destination 201.165.4.2
!
interface GigabitEthernet0/0
 description WAN_TO_ISP-R1
 ip address 201.165.5.1 255.255.255.252
 ip nat outside
 crypto map VPN
!
interface GigabitEthernet0/1
 description P2P_TO_HN-SwCore1_Gi1/0/1
 ip address 10.1.255.1 255.255.255.252
 ip ospf network point-to-point
 ip nat inside
!
interface GigabitEthernet0/2
 description P2P_TO_HN-SwCore_Gi1/0/2
 ip address 10.1.255.5 255.255.255.252
 ip ospf network point-to-point
 ip nat inside
!
router eigrp 10
 redistribute ospf 10 metric 100000 100 255 1 1500
 network 10.255.255.0 0.0.0.3
!
router ospf 10
 router-id 10.1.0.1
 redistribute eigrp 10 subnets
 network 10.1.0.1 0.0.0.0 area 0
 network 10.1.255.0 0.0.0.3 area 0
 network 10.1.255.4 0.0.0.3 area 0
 default-information originate
!
ip nat inside source list 1 interface GigabitEthernet0/0 overload
ip route 0.0.0.0 0.0.0.0 201.165.5.2
ip route 201.165.4.2 255.255.255.255 201.165.5.2
access-list 1 permit 10.1.0.0 0.0.255.255
access-list 110 permit gre host 201.165.5.1 host 201.165.4.2
end
write memory

!=====================================================================
! THIẾT BỊ: HN-SwCore1  (Multilayer Switch 3650 - Core 1 (HSRP active VLAN 10, 99; DHCP server))
!=====================================================================
enable
configure terminal
!
hostname HN-SwCore1
vlan 10
 name PC_HN_1
vlan 20
 name PC_HN_2
vlan 99
 name MGMT
!
ip dhcp excluded-address 10.1.10.1 10.1.10.10
ip dhcp excluded-address 10.1.20.1 10.1.20.10
ip dhcp pool VLAN10
 network 10.1.10.0 255.255.255.0
 default-router 10.1.10.1
 dns-server 8.8.8.8
!
ip dhcp pool VLAN20
 network 10.1.20.0 255.255.255.0
 default-router 10.1.20.1
 dns-server 8.8.8.8
!
ip routing
no ip domain-lookup
spanning-tree mode rapid-pvst
spanning-tree vlan 10,99 priority 24576
spanning-tree vlan 20 priority 28672
interface Loopback0
 ip address 10.1.0.3 255.255.255.255
!
interface Port-channel1
 switchport trunk allowed vlan 10,20,99
 switchport mode trunk
!
interface GigabitEthernet1/0/1
 no switchport
 ip address 10.1.255.2 255.255.255.252
 ip ospf network point-to-point
!
interface GigabitEthernet1/0/2
 no switchport
 ip address 10.1.255.10 255.255.255.252
 ip ospf network point-to-point
!
interface range Gi1/0/3 - 4
 description ETHERCHANNEL_TO_HN-SwCore
 switchport trunk allowed vlan 10,20,99
 switchport mode trunk
 channel-group 1 mode active
!
interface range Gi1/0/5 - 6
 description TRUNK_TO_ACCESS
 switchport trunk allowed vlan 10,20,99
 switchport mode trunk
!
interface Vlan10
 ip address 10.1.10.2 255.255.255.0
 standby 10 ip 10.1.10.1
 standby 10 priority 110
 standby 10 preempt
!
interface Vlan20
 ip address 10.1.20.2 255.255.255.0

 standby 20 priority 90
 standby 20 preempt
!
interface Vlan99

 passive-interface Vlan20
 passive-interface Vlan99

 network 10.1.255.0 0.0.0.3 area 0
